# Exact versions of every runtime package, including transitive ones, so a fresh install
# (Docker build, new venv) resolves to the same tree that has been tested instead of the
# newest release of each sub-dependency. Used together with requirements.txt:
#
#     pip install -r requirements.txt -c constraints.txt
#
# Direct dependencies are still declared and pinned in requirements.txt. This file only
# constrains what pip may pick; it does not add packages.
#
# Regenerate after changing requirements.txt, from a clean venv that has only the runtime
# packages installed (dev tools from requirements-dev.txt would leak in otherwise):
#
#     python -m venv /tmp/lockvenv && /tmp/lockvenv/bin/pip install -r requirements.txt
#     /tmp/lockvenv/bin/pip freeze | sort -f > constraints.txt   # then re-add this header
#
